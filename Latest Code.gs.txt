function getUserDataV2(uid) {
  try {
    var url = 'https://api.torn.com/v2/user/' + uid +
      '?selections=profile,battlestats,inventory&key=' + TORN_MAIN_KEY;
    var res = UrlFetchApp.fetch(url, {muteHttpExceptions: true});
    return JSON.parse(res.getContentText());
  } catch(e) { return {error: "USER_API_FAIL"}; }
}

function getFactionDataV2(fid) {
  try {
    var url = 'https://api.torn.com/v2/faction/' + fid +
      '?selections=basic&key=' + TORN_MAIN_KEY;
    var res = UrlFetchApp.fetch(url, {muteHttpExceptions: true});

}
